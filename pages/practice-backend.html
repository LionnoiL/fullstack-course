Практика: бекенд

Ці завдання виконуються локально (Node.js) або на StackBlitz із шаблоном Node/Express. Перевіряй ендпоінти через curl чи Postman.

Завдання 1. CRUD для нотаток

Побудуй Express-сервер із маршрутами для нотаток: список, отримати одну, створити, видалити (поки що зберігай у масиві в пам'яті).

Вимоги
  • GET /notes, GET /notes/:id, POST /notes, DELETE /notes/:id;
  • правильні коди статусів (201, 404, 204);
  • підключений express.json().
Рішення (фрагмент) 
const express = require('express');
const app = express();
app.use(express.json());

let notes = [];

app.get('/notes', (req, res) => res.json(notes));

app.post('/notes', (req, res) => {
  const note = { id: Date.now(), text: req.body.text };
  notes.push(note);
  res.status(201).json(note);
});

app.delete('/notes/:id', (req, res) => {
  notes = notes.filter((n) => n.id !== Number(req.params.id));
  res.status(204).end();
});

app.listen(3000);

Завдання 2. Валідація та middleware

Додай middleware логування запитів і перевірку: якщо в POST /notes немає поля text — поверни 400.

Рішення (фрагмент) 
app.use((req, res, next) => {
  console.log(`${req.method} ${req.url}`);
  next();
});

app.post('/notes', (req, res) => {
  if (!req.body.text) {
    return res.status(400).json({ error: 'Поле text обовʼязкове' });
  }
  // ...створення
});
