1. Компонент Link

Для переходів між сторінками використовують <Link> замість звичайного <a>. Він виконує клієнтську навігацію без повного перезавантаження і попередньо підвантажує сторінку.

import Link from 'next/link';

export default function Nav() {
  return (
    <nav>
      <Link href="/">Головна</Link>
      <Link href="/blog">Блог</Link>
      <Link href="/blog/hello">Стаття</Link>
    </nav>
  );
}

<Link> автоматично «префетчить» сторінки, посилання на які потрапили у видиму область — тому переходи здаються миттєвими.

2. Програмна навігація

Щоб перейти з коду (напр., після відправлення форми), беруть хук useRouter — лише у клієнтських компонентах.

'use client';
import { useRouter } from 'next/navigation';

export default function LoginForm() {
  const router = useRouter();

  function handleLogin() {
    // ...авторизація...
    router.push('/dashboard');
  }

  return <button onClick={handleLogin}>Увійти</button>;
}

3. Хуки навігації

'use client';
import { usePathname, useSearchParams, useParams } from 'next/navigation';

const pathname = usePathname();       // '/blog/hello'
const searchParams = useSearchParams(); // ?page=2
const params = useParams();           // { slug: 'hello' }

4. Підсвічування активного посилання

'use client';
import Link from 'next/link';
import { usePathname } from 'next/navigation';

export default function NavLink({ href, children }) {
  const pathname = usePathname();
  const isActive = pathname === href;
  return (
    <Link href={href} className={isActive ? 'active' : ''}>
      {children}
    </Link>
  );
}
