1. Керовані компоненти

У React значення поля прийнято зберігати у стані. Таке поле називають керованим (controlled): його value задається зі стану, а onChange оновлює стан. React стає єдиним джерелом істини.

import { useState } from 'react';

function NameField() {
  const [name, setName] = useState('');

  return (
    <input
      value={name}
      onChange={(e) => setName(e.target.value)}
    />
  );
}

2. Кілька полів одним станом

Зручно тримати всі поля в одному обʼєкті й оновлювати за атрибутом name.

function Form() {
  const [form, setForm] = useState({ email: '', password: '' });

  function handleChange(e) {
    const { name, value } = e.target;
    setForm((prev) => ({ ...prev, [name]: value }));
  }

  return (
    <>
      <input name="email" value={form.email} onChange={handleChange} />
      <input name="password" value={form.password} onChange={handleChange} />
    </>
  );
}

3. Чекбокси, select, textarea

// checkbox — беремо e.target.checked, а не value
<input type="checkbox" checked={agree} onChange={(e) => setAgree(e.target.checked)} />

// select
<select value={city} onChange={(e) => setCity(e.target.value)}>

</select>

4. Відправлення форми

function LoginForm() {
  const [form, setForm] = useState({ email: '', password: '' });
  const [error, setError] = useState('');

  function handleSubmit(e) {
    e.preventDefault();
    if (form.password.length < 8) {
      setError('Пароль закороткий');
      return;
    }
    setError('');
    console.log('Надсилаємо', form);
  }

  return (
    <form onSubmit={handleSubmit}>
      {/* поля... */}
      {error && <p>{error}</p>}
      <button>Увійти</button>
    </form>
  );
}

Для великих форм зі складною валідацією зручні бібліотеки — React Hook Form чи Formik. Вони зменшують кількість рендерів і шаблонного коду. Але спершу варто впевнено володіти керованими компонентами.
